======================================================================
4. SCRIPT DE EXCEL Y FLUJOS (crear una sola vez)
======================================================================

Dos flujos sirven para TODAS las descargas: uno genera Excel y el otro PDF.
Los dos usan el mismo script y reciben los mismos 4 datos desde la app.

----------------------------------------------------------------------
4.1 Preparar la plantilla
----------------------------------------------------------------------
1. En el sitio de SharePoint de la app: Documentos > + Nuevo > Carpeta >
   "Exportaciones".
2. Dentro de esa carpeta: + Nuevo > Libro de Excel. Ciérralo sin escribir nada
   y renómbralo "PlantillaVacia.xlsx".
3. En tu OneDrive (misma cuenta con la que crearás los flujos), crea la carpeta
   "Exportaciones". El flujo PDF guarda ahí archivos temporales y los borra al
   terminar.

----------------------------------------------------------------------
4.2 Crear el script ExportarExcelGenerico
----------------------------------------------------------------------
1. Abre PlantillaVacia.xlsx en Excel para la web.
2. Pestaña Automatizar > Nuevo script.
3. Borra el código de ejemplo y pega este:

function main(
  workbook: ExcelScript.Workbook,
  nombreHoja: string,
  columnas: string,
  datosJson: string
) {
  const datos = JSON.parse(datosJson || "[]") as Fila[];
  const hojaNombre = (nombreHoja || "Datos").replace(/[\\\/\?\*\[\]:]/g, "-").substring(0, 31);
  let hoja = workbook.getWorksheet(hojaNombre);
  if (!hoja) {
    hoja = workbook.addWorksheet(hojaNombre);
  }
  hoja.getRange().clear();

  let encabezados: string[] = (columnas || "")
    .split(";")
    .map((c) => c.trim())
    .filter((c) => c.length > 0);
  if (encabezados.length === 0 && datos.length > 0) {
    encabezados = Object.keys(datos[0]);
  }
  if (datos.length === 0 || encabezados.length === 0) {
    hoja.getRange("A1").setValue("Sin datos para el filtro seleccionado");
    hoja.activate();
    return;
  }

  hoja.getRangeByIndexes(0, 0, 1, encabezados.length).setValues([encabezados]);

  const bloque = 1000;
  for (let i = 0; i < datos.length; i += bloque) {
    const parte = datos.slice(i, i + bloque);
    const valores: (string | number | boolean)[][] = parte.map((f) =>
      encabezados.map((c) => {
        const v = f[c];
        return v === null || v === undefined ? "" : v;
      })
    );
    hoja.getRangeByIndexes(i + 1, 0, valores.length, encabezados.length).setValues(valores);
  }

  const rango = hoja.getRangeByIndexes(0, 0, datos.length + 1, encabezados.length);
  const tabla = hoja.addTable(rango, true);
  tabla.setPredefinedTableStyle("TableStyleMedium2");
  rango.getFormat().autofitColumns();
  hoja.getFreezePanes().freezeRows(1);

  const pagina = hoja.getPageLayout();
  try { pagina.setOrientation(ExcelScript.PageOrientation.landscape); } catch (e) { }
  try { pagina.setZoom({ horizontalFitToPages: 1, verticalFitToPages: 0 }); } catch (e) {
    try { pagina.setZoom({ scale: 60 }); } catch (e2) { }
  }
  try { pagina.setCenterHorizontally(true); } catch (e) { }
  try { pagina.setPrintTitleRows("$1:$1"); } catch (e) { }

  hoja.activate();
  workbook.getWorksheets().forEach((h) => {
    if (h.getName() !== hojaNombre && h.getUsedRange() === undefined) {
      h.delete();
    }
  });
}
