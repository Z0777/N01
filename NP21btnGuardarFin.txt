 CORRECCIÓN DEL ERROR DE "Patch" EN LOTE (Table / Se esperaba Record)
 btnGuardarFin y los demás botones que guardan varios registros a la vez
 (09/11/2026)
═══════════════════════════════════════════════════════════════════════════════
 Sintaxis en español: ";" separa argumentos, ";;" encadena, coma decimal.
 Fórmulas SIN "=" al inicio. Cada OnSelect va COMPLETO para copiar y pegar.

 ¿POR QUÉ SALE EL ERROR?
 ───────────────────────
 No es porque ID sea de solo lectura: el ID solo se usa para encontrar la
 fila, no se escribe.
 El problema es la forma de Patch "en tabla":
     Patch(Lista; TablaDeIDs; TablaDeCambios)
 Con esa forma, Power Apps exige que la tabla de cambios tenga EXACTAMENTE
 los tipos de la lista. colFinGuardar es una colección y guarda Estado como
 un registro de texto {Value: "Finalizado"}, no como columna de opción de
 SharePoint. Al no coincidir los tipos, Power Apps descarta la forma "tabla"
 y prueba la forma "registro" → por eso dice "Table… se esperaba Record".
 (Las versiones recientes de Power Apps revisan los tipos con más rigor; por
 eso antes podía funcionar y ahora no.)

 LA SOLUCIÓN
 ───────────
 Guardar registro por registro dentro de ForAll, con el Patch normal de un
 registro (el mismo que ya funciona en btnGuardarInd):
     ForAll(colección As _x; Patch(Lista; {ID: _x.ID}; { cambios }))
 Esta forma sí convierte {Value: "…"} a la columna de opción y no falla por
 tipos. ForAll manda los Patch en paralelo, así que sigue siendo rápido.

 ¿DÓNDE MÁS AFECTA? (todos los que guardan "en lote")
 ────────────────────────────────────────────────────
   scrRendimientos  btnGuardarFin          ← el del error
   scrRendimientos  btnMantAplicarSets     (Mantenimiento · 2A, misma forma)
   scrRendimientos  btnMantAplicarCod      (Mantenimiento · 2B, misma forma)
   Personal         btnEstadoConfirmar     (activo / inactivo en lote)
   Personal         btnHERealGuardar       (horas reales en lote)
   Personal         btnEnviarSolicitud155  (solicitudes de HE, usa Collect)
   REGISTROS        btnGuardarVisita       (gafetes y visitantes en grupo)
 Los que guardan un solo registro (btnGuardarInd, btnGuardarInicio,
 btnSalidaVis, btnEjecutarScan, btnGuardarPermiso, etc.) están correctos.

 4. Personal · btnHERealGuardar (Classic/Button) · conPopHEReal > conHERealBtns
═══════════════════════════════════════════════════════════════════════════════
 OnSelect (REEMPLAZAR COMPLETO):
With(
    {
        _i:     Value(Left(txtHEIniReal.Text; 2)) * 60 + Value(Right(txtHEIniReal.Text; 2));
        _f:     Value(Left(txtHEFinReal.Text; 2)) * 60 + Value(Right(txtHEFinReal.Text; 2));
        _usr:   User().FullName;
        _ahora: Now()
    };
    With(
        { _hr: Round(Mod(_f - _i + 1440; 1440) / 60; 2) };
        If(
            Not(IsMatch(txtHEIniReal.Text; "\d{2}:\d{2}") And IsMatch(txtHEFinReal.Text; "\d{2}:\d{2}"));
            Notify("⚠️ Escribe las horas en formato hh:mm (ej. 17:30)."; NotificationType.Error);

            Not(fxPuede("he_autorizar")) And CountRows(Filter(colHERealSel; _hr > HorasSolicitadas + 0,01)) > 0;
            Notify("⚠️ " & _hr & " h supera lo autorizado en " & CountRows(Filter(colHERealSel; _hr > HorasSolicitadas + 0,01)) & " solicitud(es). Solo el encargado puede guardarlo."; NotificationType.Error);

            ClearCollect(
                colHERealCalc;
                ForAll(colHERealSel As _s; { ID: _s.ID; Codigo: _s.Codigo; Delta: _hr - Coalesce(_s.HorasReales; _s.HorasSolicitadas) })
            );;
            Set(gErrHR; "");;
            IfError(
                ForAll(
                    colHERealSel As _s;
                    Patch(
                        SOLICITUDES_HORAS_EXTRAS;
                        { ID: _s.ID };
                        { HoraInicioReal: txtHEIniReal.Text; HoraFinReal: txtHEFinReal.Text; HorasReales: _hr; RegistroRealPor: _usr; FechaRegistroReal: _ahora }
                    )
                );
                Set(gErrHR; FirstError.Message)
            );;
            If(
                gErrHR <> "";
                Notify("❌ No se pudieron guardar todas las horas reales: " & gErrHR & " · No se tocó el acumulado; revisa y vuelve a guardar."; NotificationType.Error);;
                ClearCollect(colSolicitudesHE; Filter(SOLICITUDES_HORAS_EXTRAS; FechaRequerida >= DateAdd(Today(); -120; TimeUnit.Days)));

                ForAll(
                    GroupBy(colHERealCalc; 'Codigo'; 'Filas') As _g;
                    With(
                        { _acum: LookUp(BD_HORAS_EXTRAS_ACUMULADAS; Codigo = _g.Codigo) };
                        If(
                            !IsBlank(_acum) And Sum(_g.Filas; Delta) <> 0;
                            Patch(BD_HORAS_EXTRAS_ACUMULADAS; _acum; { TotalHorasAutorizadas: Coalesce(_acum.TotalHorasAutorizadas; 0) + Sum(_g.Filas; Delta) })
                        )
                    )
                );;
                Concurrent(
                    ClearCollect(colSolicitudesHE; Filter(SOLICITUDES_HORAS_EXTRAS; FechaRequerida >= DateAdd(Today(); -120; TimeUnit.Days)));
                    ClearCollect(colAcumuladosHE; BD_HORAS_EXTRAS_ACUMULADAS)
                );;
                Notify("✅ Horas reales guardadas: " & Text(_hr; "[$-es]0,00") & " h en " & CountRows(colHERealSel) & " solicitud(es)."; NotificationType.Success);;
                Clear(colHERealSel);;
                Clear(colHERealCalc);;
                Set(gHERealVerSel; false);;
                Set(varPopHEReal; false)
            )
        )
    )
)

                            BD_HORAS_EXTRAS_ACUMULADAS;

    Collect(colVisGrupo; { Codigo: Upper(Trim(drpVisGafete.Selected.Title)); Nombre: Upper(Trim(txtVisNombre.Text)); DPI: Trim(txtVisDPI.Text) })
);;
If(
    IsEmpty(colVisGrupo);
    Notify("⚠️ Elige el gafete y escribe el nombre del visitante."; NotificationType.Error);
    IsBlank(drpVisMotivo.Selected.Value);
    Notify("⚠️ Selecciona el motivo de la visita."; NotificationType.Error);
    CountRows(Filter(colVisGrupo; Codigo in ForAll(colAdentro; Upper(Trim(CODIGO))))) > 0;
    Notify("⚠️ Un gafete de la lista ya está en uso adentro. Quítalo de la lista."; NotificationType.Error);
    Set(gVisGrupoId; If(CountRows(colVisGrupo) > 1; "G" & Text(Now(); "yymmddhhmmss"); ""));;
    Set(gVisHasta; Coalesce(dpVisHasta.SelectedDate; Today()));;
    Set(gAhoraVis; Now());;
    Set(gVisMotivoTxt; drpVisMotivo.Selected.Value & If(IsBlank(Trim(txtVisMotivo.Text)); ""; " · " & Trim(txtVisMotivo.Text)));;
    Set(gErrVis; "");;
    IfError(
        ForAll(
            colVisGrupo As _v;
            Patch(
                GAFETES_VISITA;
                { ID: LookUp(colGafetes; Upper(Trim(Title)) = _v.Codigo).ID };
                {
                    Estado:          "Asignado";
                    Nombre:          _v.Nombre;
                    DPI:             _v.DPI;
                    Empresa:         Trim(txtVisEmpresa.Text);
                    VisitaA:         Trim(txtVisA.Text);
                    Motivo:          gVisMotivoTxt;
                    Placa:           Upper(Trim(txtVisPlaca.Text));
                    VigenteHasta:    gVisHasta;
                    GrupoVisita:     gVisGrupoId;
                    AsignadoPor:     gCorreoUsuario;
                    FechaAsignacion: gAhoraVis
                }
            )
        );
        Set(gErrVis; FirstError.Message)
    );;
    If(
        gErrVis = "";
        IfError(
            ForAll(
                colVisGrupo As _v;
                Patch(
                    REGIST;
                    Defaults(REGIST);
                    {
                        Title:            _v.Codigo & " - " & Text(Today(); "yyyymmdd");
                        FECHA:            Today();
                        CODIGO:           _v.Codigo;
                        NOMBRE:           _v.Nombre;
                        TIPO:             "Visitante";
                        AREA:             {Value: gSedeGarita};
                        DEPARTAMENTO:     "VISITANTE";
                        CATEGORIA:        {Value: "V"};
                        INGRESO:          gAhoraVis;
                        ESTADO:           {Value: "ADENTRO"};
                        DIA_LABORADO:     false;
                        VISITA_EMPRESA:   Trim(txtVisEmpresa.Text);
                        VISITA_A:         Trim(txtVisA.Text);
                        VISITA_MOTIVO:    gVisMotivoTxt;
                        VISITA_PLACA:     Upper(Trim(txtVisPlaca.Text));
                        VISITA_DPI:       _v.DPI;
                        VISITA_GRUPO:     gVisGrupoId;
                        REGISTRA_INGRESO: gCorreoUsuario
                    }
                )
            );
            Set(gErrVis; FirstError.Message)
        )
    );;
    If(
        gErrVis <> "";
        Notify("❌ No se pudo registrar: " & gErrVis; NotificationType.Error);
        Concurrent(
            ClearCollect(colGafetes; GAFETES_VISITA);
            ClearCollect(colAdentro; Filter(REGIST; ESTADO.Value = "ADENTRO"));
            ClearCollect(colHoyGar; Filter(REGIST; FECHA = Today()))
        );;
        Set(
            gUltimoScan;
            {
                Nombre:  If(CountRows(colVisGrupo) > 1; CountRows(colVisGrupo) & " visitantes"; First(colVisGrupo).Nombre);
                Tipo:    "INGRESO";
                Hora:    Now();
                Hex:     "#2563EB";
                Mensaje: "Gafete(s) " & Concat(colVisGrupo; Codigo; ", ") & " · " & drpVisMotivo.Selected.Value & " · hasta " & Text(gVisHasta; "dd/mm")
            }
        );;
        Clear(colVisGrupo);;
        Set(varPopVisita; false);;
        Set(gBusyScan; false);;
        Reset(drpVisGafete);; Reset(txtVisNombre);; Reset(txtVisDPI);; Reset(txtVisEmpresa);; Reset(txtVisA);; Reset(txtVisMotivo);; Reset(txtVisPlaca);; Reset(drpVisMotivo);; Reset(dpVisHasta);;
        SetFocus(txtScan)
    )
)
